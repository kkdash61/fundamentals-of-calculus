% ── Chapter Import Helper ────────────────────────────────────
%  \includeChapter{filename}
%    - filename: the .tex file name WITHOUT extension,
%      located inside the  chapters/  subfolder.
%    - Uses \subimport so that any \input / \includegraphics
%      inside the chapter file resolves paths relative to
%      chapters/, not to this root directory.
%
%  Each chapter file should begin directly with its content
%  (e.g. \chapter{Title of Chapter}) — no \documentclass or
%  \begin{document} in the chapter files.
% ────────────────────────────────────────────────────────────
\newcommand{\includeChapter}[1]{%
  \subimport{chapters/}{#1}%
  \clearpage                  % Start next chapter on a new page
}
